```sql
-- Run this in Supabase SQL Editor
-- WARNING: This wipes all existing data

drop table if exists achievements cascade;
drop table if exists user_kingdoms cascade;
drop table if exists kingdoms cascade;
drop table if exists users cascade;
drop table if exists workspaces cascade;

-- Workspaces
create table workspaces (
  id uuid primary key default gen_random_uuid(),
  invite_code text unique not null,
  invite_used boolean not null default false,
  achievement_threshold integer not null default 10 check (achievement_threshold between 1 and 50),
  created_at timestamptz not null default now()
);

-- Users
create table users (
  id uuid primary key default gen_random_uuid(),
  workspace_id uuid not null references workspaces(id) on delete cascade,
  display_name text not null,
  email text unique not null,
  password_hash text not null,
  created_at timestamptz not null default now()
);

-- Kingdoms (7 rows seeded separately)
create table kingdoms (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  meaning text not null,
  "order" integer not null unique check ("order" between 1 and 7),
  accent_color text not null,
  glow_color text not null,
  lore_quote text not null,
  scope text not null default 'personal' check (scope in ('personal', 'shared'))
);

-- Per-user kingdom unlock tracking (personal kingdoms only)
create table user_kingdoms (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references users(id) on delete cascade,
  workspace_id uuid not null references workspaces(id) on delete cascade,
  kingdom_id uuid not null references kingdoms(id) on delete cascade,
  unlocked_at timestamptz,
  unique(user_id, kingdom_id)
);

-- Achievements
create table achievements (
  id uuid primary key default gen_random_uuid(),
  workspace_id uuid not null references workspaces(id) on delete cascade,
  kingdom_id uuid not null references kingdoms(id),
  owner_user_id uuid references users(id),  -- null = shared kingdom achievement
  title text not null,
  note text,
  created_by uuid not null references users(id),
  completed_by uuid references users(id),
  completed_at timestamptz,
  created_at timestamptz not null default now()
);

-- Indexes
create index achievements_scoped_idx
  on achievements(workspace_id, owner_user_id, kingdom_id, completed_at);

create index user_kingdoms_user_idx
  on user_kingdoms(user_id, workspace_id);

-- Enable Realtime
alter publication supabase_realtime add table achievements;
alter publication supabase_realtime add table user_kingdoms;

-- RPC: atomic personal achievement completion (checks ownership via owner_user_id)
create or replace function complete_achievement(
  p_id uuid,
  p_user_id uuid,
  p_workspace_id uuid
)
returns setof achievements
language sql
as $$
  update achievements
  set completed_by = p_user_id, completed_at = now()
  where id = p_id
    and workspace_id = p_workspace_id
    and completed_by is null
    and (owner_user_id = p_user_id or owner_user_id is null)
  returning *;
$$;
```

Replace entire `backend/src/routes/kingdoms.js`:

```js
import { Router } from 'express'
import { supabase } from '../db/supabase.js'
import { requireAuth } from '../middleware/auth.js'

export const kingdomsRouter = Router()

// GET /kingdoms
// Returns: { myKingdoms: [...], partnerKingdoms: [...], sharedKingdom: {...} }
// myKingdoms and partnerKingdoms: 6 personal kingdoms with per-user unlock/progress
// sharedKingdom: the 7th kingdom, always active, with shared achievement count
kingdomsRouter.get('/', requireAuth, async (req, res) => {
  const { workspaceId, userId } = req.user

  const { data: workspace } = await supabase
    .from('workspaces')
    .select('achievement_threshold')
    .eq('id', workspaceId)
    .single()
  const threshold = workspace?.achievement_threshold ?? 10

  // Get both users in this workspace
  const { data: wsUsers } = await supabase
    .from('users')
    .select('id, display_name')
    .eq('workspace_id', workspaceId)
  
  const me = wsUsers?.find((u) => u.id === userId)
  const partner = wsUsers?.find((u) => u.id !== userId)

  // Get all 7 kingdoms
  const { data: kingdoms } = await supabase
    .from('kingdoms')
    .select('*')
    .order('order')

  const personalKingdoms = kingdoms?.filter((k) => k.scope === 'personal') || []
  const sharedKingdom = kingdoms?.find((k) => k.scope === 'shared')

  // Get user_kingdoms for both users
  const userIds = wsUsers?.map((u) => u.id) || []
  const { data: userKingdomRows } = await supabase
    .from('user_kingdoms')
    .select('*')
    .in('user_id', userIds)
    .eq('workspace_id', workspaceId)

  // Get completed achievement counts per (user, kingdom) for personal kingdoms
  const { data: personalCompleted } = await supabase
    .from('achievements')
    .select('kingdom_id, owner_user_id')
    .eq('workspace_id', workspaceId)
    .not('completed_at', 'is', null)
    .not('owner_user_id', 'is', null)

  // Get completed achievement count for shared kingdom
  const { data: sharedCompleted } = await supabase
    .from('achievements')
    .select('kingdom_id')
    .eq('workspace_id', workspaceId)
    .eq('kingdom_id', sharedKingdom?.id)
    .not('completed_at', 'is', null)
    .is('owner_user_id', null)

  // Build count maps
  const personalCountMap = {}
  for (const row of personalCompleted || []) {
    const key = `${row.owner_user_id}:${row.kingdom_id}`
    personalCountMap[key] = (personalCountMap[key] || 0) + 1
  }

  function buildKingdomsForUser(targetUserId) {
    return personalKingdoms.map((k) => {
      const uk = userKingdomRows?.find((r) => r.user_id === targetUserId && r.kingdom_id === k.id)
      const isUnlocked = !!uk?.unlocked_at
      const completedCount = personalCountMap[`${targetUserId}:${k.id}`] || 0
      const isDone = completedCount >= threshold

      let status
      if (!isUnlocked) status = 'locked'
      else if (isDone) status = 'completed'
      else status = 'active'

      return { ...k, completed_count: completedCount, threshold, status }
    })
  }

  const myKingdoms = buildKingdomsForUser(userId)
  const partnerKingdoms = partner ? buildKingdomsForUser(partner.id) : []

  const sharedResult = sharedKingdom
    ? {
        ...sharedKingdom,
        completed_count: sharedCompleted?.length || 0,
        threshold: null,
        status: 'active',
      }
    : null

  res.json({
    myKingdoms,
    partnerKingdoms,
    sharedKingdom: sharedResult,
    partner: partner || null,
  })
})
```

Replace the entire contents of `backend/src/routes/auth.js`:

```js
import { Router } from 'express'
import bcrypt from 'bcryptjs'
import jwt from 'jsonwebtoken'
import { v4 as uuidv4 } from 'uuid'
import { supabase } from '../db/supabase.js'

export const authRouter = Router()

async function seedUserKingdoms(userId, workspaceId) {
  const { data: kingdoms } = await supabase
    .from('kingdoms')
    .select('id, order')
    .eq('scope', 'personal')
    .order('order')

  if (!kingdoms || kingdoms.length === 0) return

  const rows = kingdoms.map((k) => ({
    user_id: userId,
    workspace_id: workspaceId,
    kingdom_id: k.id,
    unlocked_at: k.order === 1 ? new Date().toISOString() : null,
  }))

  await supabase.from('user_kingdoms').insert(rows)
}

// POST /auth/signup
authRouter.post('/signup', async (req, res) => {
  const { display_name, email, password, achievement_threshold = 10 } = req.body
  if (!display_name || !email || !password) {
    return res.status(400).json({ error: 'display_name, email, password required' })
  }
  const threshold = Math.max(1, Math.min(50, parseInt(achievement_threshold) || 10))
  const invite_code = uuidv4().slice(0, 8).toUpperCase()
  const password_hash = await bcrypt.hash(password, 10)

  const { data: workspace, error: wsErr } = await supabase
    .from('workspaces')
    .insert({ invite_code, achievement_threshold: threshold })
    .select()
    .single()
  if (wsErr) return res.status(500).json({ error: wsErr.message })

  const { data: user, error: userErr } = await supabase
    .from('users')
    .insert({ workspace_id: workspace.id, display_name, email, password_hash })
    .select('id, display_name, email, workspace_id')
    .single()
  if (userErr) return res.status(500).json({ error: userErr.message })

  await seedUserKingdoms(user.id, workspace.id)

  const token = jwt.sign(
    { userId: user.id, workspaceId: workspace.id },
    process.env.JWT_SECRET,
    { expiresIn: '30d' }
  )
  res.json({ token, user, invite_code })
})

// POST /auth/join
authRouter.post('/join', async (req, res) => {
  const { display_name, email, password, invite_code } = req.body
  if (!display_name || !email || !password || !invite_code) {
    return res.status(400).json({ error: 'display_name, email, password, invite_code required' })
  }

  const { data: workspace, error: wsErr } = await supabase
    .from('workspaces')
    .select('*')
    .eq('invite_code', invite_code.toUpperCase())
    .single()
  if (wsErr || !workspace) return res.status(404).json({ error: 'Invite code not found' })
  if (workspace.invite_used) return res.status(400).json({ error: 'Invite code already used' })

  const { data: claimed, error: claimErr } = await supabase
    .from('workspaces')
    .update({ invite_used: true })
    .eq('id', workspace.id)
    .eq('invite_used', false)
    .select('id')

  if (claimErr || !claimed || claimed.length === 0) {
    return res.status(400).json({ error: 'Invite code already used' })
  }

  const password_hash = await bcrypt.hash(password, 10)
  const { data: user, error: userErr } = await supabase
    .from('users')
    .insert({ workspace_id: workspace.id, display_name, email, password_hash })
    .select('id, display_name, email, workspace_id')
    .single()
  if (userErr) return res.status(500).json({ error: userErr.message })

  await seedUserKingdoms(user.id, workspace.id)

  const token = jwt.sign(
    { userId: user.id, workspaceId: workspace.id },
    process.env.JWT_SECRET,
    { expiresIn: '30d' }
  )
  res.json({ token, user })
})

// POST /auth/login
authRouter.post('/login', async (req, res) => {
  const { email, password } = req.body
  if (!email || !password) return res.status(400).json({ error: 'email, password required' })

  const { data: user, error } = await supabase
    .from('users')
    .select('*')
    .eq('email', email)
    .single()
  if (error || !user) return res.status(401).json({ error: 'Invalid credentials' })

  const valid = await bcrypt.compare(password, user.password_hash)
  if (!valid) return res.status(401).json({ error: 'Invalid credentials' })

  const token = jwt.sign(
    { userId: user.id, workspaceId: user.workspace_id },
    process.env.JWT_SECRET,
    { expiresIn: '30d' }
  )
  const { password_hash, ...safeUser } = user
  res.json({ token, user: safeUser })
})
```

- [ ] **Step 1: Rewrite achievements route with owner_user_id, scoped fetch, and per-user unlock**

Replace entire `backend/src/routes/achievements.js`:

```js
import { Router } from 'express'
import { supabase } from '../db/supabase.js'
import { requireAuth } from '../middleware/auth.js'

export const achievementsRouter = Router()

async function broadcastKingdomUnlocked(workspaceId, payload) {
  const url = `${process.env.SUPABASE_URL}/realtime/v1/api/broadcast`
  await fetch(url, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${process.env.SUPABASE_SERVICE_ROLE_KEY}`,
    },
    body: JSON.stringify({
      messages: [{ topic: `workspace:${workspaceId}`, event: 'kingdom_unlocked', payload }],
    }),
  })
}

// GET /achievements?kingdom_id=&scope=mine|partner|shared
// scope=mine: owner_user_id = me (personal kingdom)
// scope=partner: owner_user_id = partner (their personal kingdom, read-only)
// scope=shared: owner_user_id IS NULL (Miveritas)
achievementsRouter.get('/', requireAuth, async (req, res) => {
  const { workspaceId, userId } = req.user
  const { kingdom_id, scope } = req.query

  let query = supabase
    .from('achievements')
    .select('*, creator:created_by(id,display_name), completer:completed_by(id,display_name)')
    .eq('workspace_id', workspaceId)
    .order('created_at', { ascending: false })

  if (kingdom_id) query = query.eq('kingdom_id', kingdom_id)

  if (scope === 'mine') {
    query = query.eq('owner_user_id', userId)
  } else if (scope === 'shared') {
    query = query.is('owner_user_id', null)
  } else if (scope === 'partner') {
    // Get partner id
    const { data: wsUsers } = await supabase
      .from('users')
      .select('id')
      .eq('workspace_id', workspaceId)
    const partnerId = wsUsers?.find((u) => u.id !== userId)?.id
    if (partnerId) query = query.eq('owner_user_id', partnerId)
    else return res.json([])
  }

  const { data, error } = await query
  if (error) return res.status(500).json({ error: error.message })
  res.json(data)
})

// POST /achievements
// For personal kingdoms: owner_user_id = userId
// For shared kingdom: owner_user_id = null
achievementsRouter.post('/', requireAuth, async (req, res) => {
  const { workspaceId, userId } = req.user
  const { kingdom_id, title, note, is_shared } = req.body
  if (!kingdom_id || !title?.trim()) {
    return res.status(400).json({ error: 'kingdom_id and title required' })
  }

  const owner_user_id = is_shared ? null : userId

  const { data, error } = await supabase
    .from('achievements')
    .insert({
      workspace_id: workspaceId,
      kingdom_id,
      owner_user_id,
      title: title.trim(),
      note: note || null,
      created_by: userId,
    })
    .select('*, creator:created_by(id,display_name)')
    .single()
  if (error) return res.status(500).json({ error: error.message })
  res.status(201).json(data)
})

// PATCH /achievements/:id/complete
// Allowed: owner (personal), either user (shared)
achievementsRouter.patch('/:id/complete', requireAuth, async (req, res) => {
  const { workspaceId, userId } = req.user
  const { id } = req.params

  const { data, error } = await supabase.rpc('complete_achievement', {
    p_id: id,
    p_user_id: userId,
    p_workspace_id: workspaceId,
  })

  if (error) return res.status(500).json({ error: error.message })
  if (!data || data.length === 0) {
    const { data: existing } = await supabase
      .from('achievements')
      .select('*, completer:completed_by(id,display_name)')
      .eq('id', id)
      .single()
    return res.status(409).json({ error: 'Already completed', achievement: existing })
  }

  const achievement = data[0]
  const isShared = achievement.owner_user_id === null

  const { data: workspace } = await supabase
    .from('workspaces')
    .select('achievement_threshold')
    .eq('id', workspaceId)
    .single()
  const threshold = workspace?.achievement_threshold ?? 10

  if (isShared) {
    // Shared kingdom: no unlock logic, just return
    return res.json({ achievement, newKingdomUnlocked: false, journeyComplete: false, kingdom: null })
  }

  // Personal kingdom: check per-user unlock
  const { count } = await supabase
    .from('achievements')
    .select('*', { count: 'exact', head: true })
    .eq('workspace_id', workspaceId)
    .eq('kingdom_id', achievement.kingdom_id)
    .eq('owner_user_id', userId)
    .not('completed_at', 'is', null)

  let newKingdomUnlocked = false
  let journeyComplete = false
  let unlockedKingdom = null

  if (count >= threshold) {
    const { data: currentKingdom } = await supabase
      .from('kingdoms')
      .select('order')
      .eq('id', achievement.kingdom_id)
      .single()

    // Find next personal kingdom not yet unlocked for this user
    const { data: allPersonalKingdoms } = await supabase
      .from('kingdoms')
      .select('*')
      .eq('scope', 'personal')
      .gt('order', currentKingdom.order)
      .order('order')
      .limit(1)

    if (allPersonalKingdoms && allPersonalKingdoms.length > 0) {
      const nextKingdom = allPersonalKingdoms[0]
      const { data: existingUK } = await supabase
        .from('user_kingdoms')
        .select('id, unlocked_at')
        .eq('user_id', userId)
        .eq('kingdom_id', nextKingdom.id)
        .single()

      if (existingUK && !existingUK.unlocked_at) {
        const now = new Date().toISOString()

          .from('user_kingdoms')
          .update({ unlocked_at: now })
          .eq('id', existingUK.id)

        await broadcastKingdomUnlocked(workspaceId, {
          kingdom: unlockedKingdom,
          userId,
        })
      }
    } else {
      // No next kingdom — check if this user completed all 6
      const { data: myUK } = await supabase
        .from('user_kingdoms')
        .select('unlocked_at')
        .eq('user_id', userId)
        .eq('workspace_id', workspaceId)
      const allUnlocked = myUK?.every((r) => !!r.unlocked_at)
      if (allUnlocked) {
        journeyComplete = true
        await broadcastKingdomUnlocked(workspaceId, { kingdom: null, journeyComplete: true, userId })
      }
    }
  }

  res.json({ achievement, newKingdomUnlocked, journeyComplete, kingdom: unlockedKingdom })
})

achievementsRouter.delete('/:id', requireAuth, async (req, res) => {
  const { userId, workspaceId } = req.user
  const { id } = req.params

  const { data: existing } = await supabase
    .from('achievements')
    .select('created_by, completed_at, owner_user_id')
    .eq('id', id)
    .eq('workspace_id', workspaceId)
    .single()

  const isShared = existing.owner_user_id === null
  if (!isShared && existing.created_by !== userId) {
    return res.status(403).json({ error: 'Not your achievement' })
  }
  if (existing.completed_at) return res.status(400).json({ error: 'Cannot delete completed achievement' })

  const { error } = await supabase.from('achievements').delete().eq('id', id)
  if (error) return res.status(500).json({ error: error.message })
```

```js
import { supabase } from '../lib/supabase.js'
import { useAppStore } from '../store/useAppStore.js'

export function useRealtimeSync(workspaceId) {
  const addAchievement = useAppStore((s) => s.addAchievement)
  const updateAchievement = useAppStore((s) => s.updateAchievement)
  const removeAchievement = useAppStore((s) => s.removeAchievement)
  const triggerUnlock = useAppStore((s) => s.triggerUnlock)
  const loadKingdoms = useAppStore((s) => s.loadKingdoms)
  const userId = useAppStore((s) => s.user?.id)

  useEffect(() => {
    if (!workspaceId) return

    const tableChannel = supabase
      .channel(`achievements:${workspaceId}`)
      .on(
        'postgres_changes',
        {
          event: 'INSERT',
          schema: 'public',
          table: 'achievements',
          filter: `workspace_id=eq.${workspaceId}`,
        },
        (payload) => addAchievement(payload.new)
      )
      .on(
        'postgres_changes',
        {
          event: 'UPDATE',
          schema: 'public',
          table: 'achievements',
          filter: `workspace_id=eq.${workspaceId}`,
        },
        (payload) => updateAchievement(payload.new)
      )
      .on(
        'postgres_changes',
        {
          event: 'DELETE',
          schema: 'public',
          table: 'achievements',
          filter: `workspace_id=eq.${workspaceId}`,
        },
        (payload) => removeAchievement(payload.old.id)
      )
      .subscribe()

    // user_kingdoms changes: reload kingdoms when partner unlocks one
    const ukChannel = supabase
      .channel(`user_kingdoms:${workspaceId}`)
      .on(
        'postgres_changes',
        {
          event: 'UPDATE',
          schema: 'public',
          table: 'user_kingdoms',
        },
        () => loadKingdoms()
      )
      .subscribe()

    const broadcastChannel = supabase
      .channel(`workspace:${workspaceId}`)
      .on('broadcast', { event: 'kingdom_unlocked' }, ({ payload }) => {
        // Only trigger cinematic for the user who completed it
        if (payload.userId && payload.userId !== userId) {
          // Partner unlocked something — just reload kingdoms silently
          loadKingdoms()
          return
        }
        triggerUnlock(payload.kingdom, payload.journeyComplete || false)
      })
      .subscribe()

    return () => {
      supabase.removeChannel(tableChannel)
      supabase.removeChannel(ukChannel)
      supabase.removeChannel(broadcastChannel)
    }
  }, [workspaceId, userId, addAchievement, updateAchievement, removeAchievement, triggerUnlock, loadKingdoms])
}
```

```jsx
      timers.push(setTimeout(() => setLitCount(i + 1), 800 + i * 200))
    })
    // After all lit, show text
    timers.push(setTimeout(() => setShowText(true), 800 + KINGDOM_ORDER.length * 200 + 400))
    // Begin fade-out
    timers.push(setTimeout(() => setFading(true), 7000))
    // Dismiss
    timers.push(setTimeout(() => clearJourneyComplete(), 8000))
    return () => timers.forEach(clearTimeout)
  }, [clearJourneyComplete])

  return (
    <div
      className="fixed inset-0 z-50 flex flex-col items-center justify-center"
      style={{
        backgroundColor: '#050D20',
        opacity: fading ? 0 : 1,
        transition: 'opacity 1000ms ease',
      }}
    >
      {/* Kingdom glow chips */}
      <div className="flex gap-3 flex-wrap justify-center mb-10">
        {KINGDOM_ORDER.map((name, i) => {
          const colors = KINGDOM_COLORS[name]
          const isLit = i < litCount
          return (
            <div
              key={name}
              className="px-4 py-2 rounded-full text-sm font-semibold"
              style={{
                fontFamily: 'Georgia, serif',
                backgroundColor: isLit ? colors.accent + '22' : '#111D45',
                color: isLit ? colors.accent : '#3a4a6a',
                border: `1px solid ${isLit ? colors.accent : '#243060'}`,
                boxShadow: isLit ? `0 0 20px ${colors.glow}44` : 'none',
                transition: 'all 400ms ease',
                letterSpacing: '0.05em',
              }}
            >
              {name}
            </div>
          )
        })}
      </div>

      {/* Final text */}
      <div
        className="text-center"
        style={{
          opacity: showText ? 1 : 0,
          transform: showText ? 'translateY(0)' : 'translateY(20px)',
          transition: 'opacity 800ms ease, transform 800ms ease',
        }}
      >
        <h1
          className="text-5xl font-bold mb-4"
          style={{
            color: '#F7F1FF',
            fontFamily: 'Georgia, serif',
            letterSpacing: '0.08em',
            textShadow: '0 0 60px #BDEBFF44',
          }}
        >
          The Journey is Complete
        </h1>
        <p className="text-lg italic" style={{ color: '#8A9BB8' }}>
          All kingdoms of the Land-of-Almost-Awake have been explored.
        </p>
      </div>
    </div>
  )
}
```
